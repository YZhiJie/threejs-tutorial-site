核心概念讲解 ​

本篇把前面几乎所有模块串成一个完整场景：雾 + 阴影 + 实例化建筑群 + 信标拾取 + 旋转雷达 + 粒子 + HUD。重点不是某个特效，而是它们如何在一个循环里协同。

场景技术清单与对应章节 ​

技术作用出处
FogExp2 指数雾统一远景、藏住场景边界2.2
PCF 阴影（r186）建筑投影8.2
InstancedMesh ×26 栋楼1 个 draw call 画建筑群4.3 / 11.3
Raycaster 信标拾取点击信标弹出信息10.1
Group 旋转雷达父子层级驱动扫描2.3
Points 粒子漂浮尘埃5.2 思路
HTML HUD状态/提示信息10.3

组织方式 ​

  • 建筑用 InstancedMesh 一次提交，颜色逐实例 setColorAt
  • 信标是少量独立 Mesh（要单独拾取/缩放动画），不进实例化
  • 雷达碟挂在一个 pivot Group 上，转 pivot 即转碟
  • 粒子是一块大的 Points，整体缓慢旋转即可，不必逐顶点动

代码示例 ​

下面是核心逻辑（完整含 HUD 的页面见 DemoFrame"复制源码"）：

js
import * as THREE from 'three'
import { OrbitControls } from 'three/addons/controls/OrbitControls.js'

/* ---------- 基础 + 雾 ---------- */
const scene = new THREE.Scene()
scene.background = new THREE.Color(0x05070c)
scene.fog = new THREE.FogExp2(0x05070c, 0.022)   // 指数雾统一远景
const camera = new THREE.PerspectiveCamera(50, innerWidth / innerHeight, 0.1, 300)
camera.position.set(24, 18, 28)
const renderer = new THREE.WebGLRenderer({ antialias: true })
renderer.setSize(innerWidth, innerHeight)
renderer.setPixelRatio(Math.min(devicePixelRatio, 2))
renderer.shadowMap.enabled = true
renderer.shadowMap.type = THREE.PCFShadowMap
document.getElementById('app').appendChild(renderer.domElement)
const controls = new OrbitControls(camera, renderer.domElement)
controls.enableDamping = true
controls.maxPolarAngle = Math.PI / 2 - 0.04

/* ---------- 灯光 ---------- */
scene.add(new THREE.HemisphereLight(0x334466, 0x05070c, 0.55))
const sun = new THREE.DirectionalLight(0x9fc0ff, 2)
sun.position.set(20, 30, 12)
sun.castShadow = true
sun.shadow.mapSize.set(2048, 2048)
Object.assign(sun.shadow.camera, { near: 1, far: 90, left: -40, right: 40, top: 40, bottom: -40 })
scene.add(sun)

/* ---------- 地面网格 ---------- */
const ground = new THREE.Mesh(
  new THREE.PlaneGeometry(160, 160),
  new THREE.MeshStandardMaterial({ color: 0x0c1320, roughness: 0.95 })
)
ground.rotation.x = -Math.PI / 2
ground.receiveShadow = true
scene.add(ground)
const grid = new THREE.GridHelper(160, 80, 0x1d3a5f, 0x122036)
grid.position.y = 0.02
scene.add(grid)

/* ---------- 26 栋实例化建筑 ---------- */
const BUILDINGS = 26
const boxGeo = new THREE.BoxGeometry(1, 1, 1)
const boxMat = new THREE.MeshStandardMaterial({ roughness: 0.55, metalness: 0.35 })
const city = new THREE.InstancedMesh(boxGeo, boxMat, BUILDINGS)
city.castShadow = city.receiveShadow = true

const dummy = new THREE.Object3D()
const col = new THREE.Color()
const buildingData = []
for (let i = 0; i < BUILDINGS; i++) {
  // 在避开中心广场的环形区域随机摆放
  let x, z
  do { x = (Math.random() - 0.5) * 44; z = (Math.random() - 0.5) * 44 }
  while (Math.hypot(x, z) < 6)
  const h = 2 + Math.random() * 8
  const w = 1.6 + Math.random() * 1.6
  dummy.position.set(x, h / 2, z)
  dummy.scale.set(w, h, w)
  dummy.rotation.y = Math.random() * Math.PI
  dummy.updateMatrix()
  city.setMatrixAt(i, dummy.matrix)
  // 冷蓝灰为主、少量暖黄亮灯楼
  const lit = Math.random() > 0.82
  city.setColorAt(i, col.setHSL(lit ? 0.12 : 0.58, lit ? 0.7 : 0.3, lit ? 0.6 : 0.25 + Math.random() * 0.15))
  buildingData.push({ x, z, h })
}
city.instanceMatrix.needsUpdate = true
city.instanceColor.needsUpdate = true
scene.add(city)

/* ---------- 3 个可拾取信标 ---------- */
const beaconMeshes = []
const beacons = [
  { name: '能量核心', pos: [0, 1.2, 0], info: '基地主反应堆，输出 4.2 TW。' },
  { name: '北塔雷达', pos: [0, 5.5, -5.2], info: '远程预警与轨道扫描阵列。' },
  { name: '停机坪',   pos: [5.2, 0.6, 2.5], info: '运输无人机起降区。' }
]
beacons.forEach(b => {
  const m = new THREE.Mesh(
    new THREE.CylinderGeometry(0.12, 0.18, 2.2, 16),
    new THREE.MeshStandardMaterial({ color: 0x22d3ee, emissive: 0x0891b2, emissiveIntensity: 1.2 })
  )
  m.position.set(...b.pos)
  m.userData = b
  scene.add(m)
  beaconMeshes.push(m)
})

/* ---------- 雷达碟：pivot Group 驱动 ---------- */
const dishPivot = new THREE.Group()
dishPivot.position.set(0, 6.4, -5.2)
const dish = new THREE.Mesh(
  new THREE.ConeGeometry(1.4, 0.5, 32, 1, true),
  new THREE.MeshStandardMaterial({ color: 0x94a3b8, roughness: 0.4, side: THREE.DoubleSide })
)
dish.rotation.x = Math.PI / 2
dishPivot.add(dish)
scene.add(dishPivot)

/* ---------- 漂浮尘埃粒子 ---------- */
const DUST = 900
const dustPos = new Float32Array(DUST * 3)
for (let i = 0; i < DUST; i++) {
  dustPos[i*3]   = (Math.random() - 0.5) * 80
  dustPos[i*3+1] = Math.random() * 24
  dustPos[i*3+2] = (Math.random() - 0.5) * 80
}
const dustGeo = new THREE.BufferGeometry()
dustGeo.setAttribute('position', new THREE.BufferAttribute(dustPos, 3))
const dust = new THREE.Points(dustGeo,
  new THREE.PointsMaterial({ color: 0x67e8f9, size: 0.12, transparent: true, opacity: 0.55 }))
scene.add(dust)

/* ---------- 信标拾取（含防拖拽误触）---------- */
const raycaster = new THREE.Raycaster()
const pointer = new THREE.Vector2()
let hoveredBeacon = null
let downX = 0, downY = 0
renderer.domElement.addEventListener('pointerdown', e => { downX = e.clientX; downY = e.clientY })
renderer.domElement.addEventListener('pointermove', e => {
  const r = renderer.domElement.getBoundingClientRect()
  pointer.set(((e.clientX - r.left) / r.width) * 2 - 1, -((e.clientY - r.top) / r.height) * 2 + 1)
  raycaster.setFromCamera(pointer, camera)
  hoveredBeacon = raycaster.intersectObjects(beaconMeshes)[0]?.object ?? null
  renderer.domElement.style.cursor = hoveredBeacon ? 'pointer' : 'grab'
})
renderer.domElement.addEventListener('click', e => {
  if (Math.hypot(e.clientX - downX, e.clientY - downY) > 5) return
  if (hoveredBeacon) showInfo(hoveredBeacon.userData)
})
function showInfo(data) {
  const hud = document.getElementById('info')
  hud.querySelector('h3').textContent = data.name
  hud.querySelector('p').textContent = data.info
  hud.classList.add('show')
}

/* ---------- 主循环：一个 dt 驱动所有动态元素 ---------- */
const params = { radarSpeed: 1 }
const clock = new THREE.Timer()
clock.connect(document) // r183 起 Clock 已弃用，改用 Timer
renderer.setAnimationLoop(() => {
  clock.update() // 每帧先 update，再读取
  const dt = clock.getDelta()
  const t = clock.getElapsed()

  dishPivot.rotation.y += dt * params.radarSpeed * 2
  dishPivot.rotation.x = Math.sin(t * 0.6) * 0.35 - 0.3
  dust.rotation.y += dt * 0.02

  // 信标呼吸缩放；被悬停的那个不缩放（交给 hover 放大逻辑）
  beaconMeshes.forEach((b, i) => {
    const s = 1 + Math.sin(t * 3 + i) * 0.18
    if (b !== hoveredBeacon) b.scale.setScalar(s)
    else b.scale.setScalar(1.35)
  })

  controls.update()
  renderer.render(scene, camera)
})

常见问题解答 ​

Q1：为什么建筑用 InstancedMesh，信标却用独立 Mesh？

建筑数量大（26→可扩到上千）且只需要整体绘制，实例化能把 draw call 压到 1；信标只有 3 个，却要被 Raycaster 单独拾取、各自做呼吸/悬停动画，独立 Mesh 更直接。按是否需要"单独控制"决定是否实例化。

Q2：粒子为什么整体旋转而不是逐粒更新？

900 个粒子每帧逐点改 position 会有 CPU 上传开销。视觉上"整片尘埃缓慢漂移"只需旋转整个 Points 对象即可满足；若要每粒独立运动再考虑逐顶点更新或 GPU 着色器。

Q3：雾把建筑吞得太狠怎么办？

调小 FogExp2 的 density（如 0.022→0.012），或改用线性 Fog 精确控制 near/far。雾色必须与背景色一致，否则边界出现灰幕。

动手小练习 ​

  1. 给建筑窗户做发光贴图（emissiveMap + Canvas 网格窗格），让亮灯楼更像城市夜景。
  2. 点击信标时相机用 10.3 的阻尼飞行聚焦过去，再点空白返回总览。
  3. 加一道绕能量核心缓慢公转的蓝色光环（TorusGeometry + 公转节点）。
