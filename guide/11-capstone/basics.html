11.1 基础入门：项目分层架构（Bootstrap） ​

核心概念讲解 ​

学到这里，你已经掌握了全部核心零件。综合实战要解决的不再是"某个 API 怎么用"，而是"怎么组织一个不会随规模膨胀而失控的项目"。本篇先把脚手架搭对。

分层架构：各司其职的五个模块 ​

这种拆法的收益：

  • 换内容不动基础设施：WorldBuilder 整个替换，Stage/Loop 一行不改
  • update 解耦：任何对象只要实现 update(dt) 就能加入主循环
  • 可测试：每层构造参数明确，能单独实例化排查

术语解释：关注点分离（Separation of Concerns） 把"怎么渲染""渲染什么""何时更新"分到不同模块，是工程上控制复杂度的核心原则。

启动顺序很重要 ​

正确顺序：创建 renderer → 场景/相机 → 控制器 → 构建世界（依赖 renderer 生成环境贴图时）→ 启动循环。顺序错了常见症状是"环境贴图是黑的""控制器不生效"。

代码示例 ​

下面是带启动加载层与日志的完整分层骨架（可运行完整版见 DemoFrame）：

js
import * as THREE from 'three'
import { OrbitControls } from 'three/addons/controls/OrbitControls.js'
import { RoomEnvironment } from 'three/addons/environments/RoomEnvironment.js'

/* 一个极简日志器：带模块前缀，便于在控制台区分来源 */
const log = (tag, ...args) => console.log(`%c[${tag}]`, 'color:#38bdf8', ...args)

/* ---------- Stage：基础设施 ---------- */
class Stage {
  constructor(container) {
    log('Stage', '初始化渲染器 / 相机 / 场景')
    this.renderer = new THREE.WebGLRenderer({ antialias: true })
    this.renderer.setSize(innerWidth, innerHeight)
    this.renderer.setPixelRatio(Math.min(devicePixelRatio, 2))
    this.renderer.shadowMap.enabled = true
    this.renderer.shadowMap.type = THREE.PCFShadowMap // r186 默认（PCFSoft 已移除）
    this.renderer.toneMapping = THREE.ACESFilmicToneMapping
    container.appendChild(this.renderer.domElement)

    this.scene = new THREE.Scene()
    this.scene.background = new THREE.Color(0x0b0e14)

    // 环境贴图必须在 renderer 存在之后生成（PMREM 依赖 WebGL 上下文）
    this.scene.environment = new THREE.PMREMGenerator(this.renderer)
      .fromScene(new RoomEnvironment(), 0.04).texture

    this.camera = new THREE.PerspectiveCamera(50, innerWidth / innerHeight, 0.1, 200)
    this.camera.position.set(8, 6, 10)

    this.controls = new OrbitControls(this.camera, this.renderer.domElement)
    this.controls.enableDamping = true
    this.controls.maxPolarAngle = Math.PI / 2 - 0.04

    this.resizer = new Resizer(this)
  }
}

/* ---------- Resizer：自适应单独成类 ---------- */
class Resizer {
  constructor(stage) {
    this.stage = stage
    addEventListener('resize', () => this.onResize())
  }
  onResize() {
    const { camera, renderer } = this.stage
    camera.aspect = innerWidth / innerHeight
    camera.updateProjectionMatrix()
    renderer.setSize(innerWidth, innerHeight)
    log('Resizer', `尺寸更新 ${innerWidth}×${innerHeight}`)
  }
}

/* ---------- WorldBuilder：只管内容 ---------- */
class WorldBuilder {
  constructor(scene) {
    this.scene = scene
    this.updatables = []   // 暴露给 Loop 的更新列表
  }
  build() {
    log('World', '构建灯光 / 地面 / 物体')
    this.scene.add(new THREE.HemisphereLight(0xbfd9ff, 0x202020, 0.6))

    const sun = new THREE.DirectionalLight(0xffffff, 2)
    sun.position.set(8, 12, 6)
    sun.castShadow = true
    sun.shadow.mapSize.set(2048, 2048)
    Object.assign(sun.shadow.camera, { near: 1, far: 40, left: -14, right: 14, top: 14, bottom: -14 })
    this.scene.add(sun)

    const ground = new THREE.Mesh(
      new THREE.PlaneGeometry(60, 60),
      new THREE.MeshStandardMaterial({ color: 0x1c2330, roughness: 1 })
    )
    ground.rotation.x = -Math.PI / 2
    ground.receiveShadow = true
    this.scene.add(ground)

    const tower = new THREE.Mesh(
      new THREE.CylinderGeometry(0.6, 0.9, 3, 24),
      new THREE.MeshStandardMaterial({ color: 0x60a5fa, roughness: 0.4 })
    )
    tower.position.y = 1.5
    tower.castShadow = tower.receiveShadow = true
    this.scene.add(tower)

    // 任何需要每帧更新的对象都注册进 updatables
    this.updatables.push({
      update(dt) { tower.rotation.y += dt * 0.4 }
    })
    return { tower }
  }
}

/* ---------- Loop：统一时钟 ---------- */
class Loop {
  constructor(stage, updatables = []) {
    this.stage = stage
    this.updatables = updatables
    this.clock = new THREE.Timer()
    this.clock.connect(document) // r183 起 Clock 已弃用，改用 Timer
  }
  start() {
    log('Loop', '启动渲染循环')
    this.stage.renderer.setAnimationLoop(() => this.tick())
  }
  tick() {
    this.clock.update() // 每帧先 update，再读取
    const dt = Math.min(this.clock.getDelta(), 0.05)  // 钳制防止切后台跳变
    for (const u of this.updatables) u.update?.(dt)
    this.stage.controls.update()
    this.stage.renderer.render(this.stage.scene, this.stage.camera)
  }
}

/* ---------- App：编排启动流程 ---------- */
class App {
  constructor(container) {
    this.stage = new Stage(container)
    this.world = new WorldBuilder(this.stage.scene)
    this.world.build()
    this.loop = new Loop(this.stage, this.world.updatables)
  }
  start() { this.loop.start(); log('App', '启动完成 ✓') }
}

// DOM 就绪后启动；真实项目可在这里加加载层淡出
{
  const loading = document.getElementById('loading')
  const app = new App(document.getElementById('app'))
  app.start()
  loading?.style.setProperty('opacity', '0')
  setTimeout(() => loading?.remove(), 400)
}

常见问题解答 ​

Q1：为什么环境贴图要在 Stage 里、而不是 WorldBuilder 里建？

PMREMGenerator 依赖已创建的 WebGL renderer，而 renderer 属于 Stage 的职责。环境贴图是"渲染基础设施"的一部分，放 Stage 更合理；WorldBuilder 只管摆灯和物体。

Q2：updatables 数组的好处是什么？

Loop 不需要知道每个对象具体是什么，只统一调用 update(dt)。加新动画（粒子、旋转、相机运镜）时只要 push 一个带 update 的对象，Loop 完全不用改，符合"对扩展开放"。

Q3：加载层什么时候真正必要？

当场景要异步加载 GLTF 模型、HDR 贴图、贴图图集时，用 LoadingManager 跟踪进度，全部就绪后再淡出加载层。纯程序化小场景可省略，但保留这个钩子不亏。

动手小练习 ​

  1. 给 Loop 加 pause()/resume()，并在暂停时不更新 updatables 但仍渲染一帧。
  2. 把 WorldBuilder 拆成 Lights、Terrain、Props 三个子 builder，由 App 依次 build。
  3. 在日志器里加上时间戳和 FPS 输出，形成你自己的项目调试模板。
