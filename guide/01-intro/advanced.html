核心概念讲解 ​

为什么要从单文件迁移到模块化 ​

单文件塞进一个 <script> 标签在入门时没问题，项目一长大就会遇到硬伤：单文件难维护、多场景无法复用、工具函数无处安放。社区的普遍做法是把一个 3D 场景拆成几个职责清晰的层：

调试三件套 ​

  1. AxesHelper —— 红 X / 绿 Y / 蓝 Z 三根坐标轴，建立空间方向感
  2. GridHelper —— 地板网格，直观感受位置与大小
  3. lil-gui —— 运行时参数面板，拖滑块立即看到画面变化

性能数据从哪里读 ​

js
// renderer.info 是只读的实时统计，每帧都可读
const info = renderer.info.render
console.log(info.calls, info.triangles) // draw calls 数 / 三角形总数

术语解释：draw call（绘制调用）是 CPU 命令 GPU 画一批三角形的次数。次数越多 CPU 开销越大，所以优化时常说"合并几何体、用 InstancedMesh 减少 draw call"。

代码示例 ​

下面是"Stage + WorldBuilder + Loop"三层架构的核心代码（含调试三件套与 FPS 统计）。完整可运行页面（含 importmap/HTML 包装）见下方 DemoFrame 的"复制源码"：

js
import * as THREE from 'three'
import { OrbitControls } from 'three/addons/controls/OrbitControls.js'
import GUI from 'three/addons/libs/lil-gui.module.min.js'

/* ============ Stage 层：只管基础设施，不知道放了什么物体 ============ */
class Stage {
  constructor(container) {
    this.renderer = new THREE.WebGLRenderer({ antialias: true })
    this.renderer.setSize(innerWidth, innerHeight)
    this.renderer.setPixelRatio(Math.min(devicePixelRatio, 2))
    this.renderer.shadowMap.enabled = true                 // 全局阴影总开关
    this.renderer.shadowMap.type = THREE.PCFShadowMap // r186 默认的 PCF 过滤（PCFSoft 已移除）
    container.appendChild(this.renderer.domElement)

    this.scene = new THREE.Scene()
    this.scene.background = new THREE.Color(0x0b0e14)

    this.camera = new THREE.PerspectiveCamera(50, innerWidth / innerHeight, 0.1, 100)
    this.camera.position.set(3, 2.4, 4)

    this.controls = new OrbitControls(this.camera, this.renderer.domElement)
    this.controls.enableDamping = true

    addEventListener('resize', () => this.resize())
  }
  resize() {
    this.camera.aspect = innerWidth / innerHeight
    this.camera.updateProjectionMatrix()
    this.renderer.setSize(innerWidth, innerHeight)
  }
}

/* ============ WorldBuilder 层：只管往 scene 里摆放内容 ============ */
class WorldBuilder {
  constructor(scene) {
    this.scene = scene
    this.updatables = []   // 收集所有"每帧要更新"的对象
  }

  build() {
    // ① 调试辅助：坐标轴 + 网格 —— 搭新场景时第一批代码
    this.scene.add(new THREE.AxesHelper(2))
    this.scene.add(new THREE.GridHelper(10, 20))

    // ② 灯光：半球光打底 + 方向光主照明
    this.scene.add(new THREE.HemisphereLight(0xbfd9ff, 0x202020, 0.5))
    const sun = new THREE.DirectionalLight(0xffffff, 1.6)
    sun.position.set(3, 5, 4)
    sun.castShadow = true                    // 光源不开 castShadow 就没阴影
    sun.shadow.mapSize.set(1024, 1024)       // 阴影贴图分辨率
    this.scene.add(sun)

    // ③ 被调试物体
    const material = new THREE.MeshStandardMaterial({ color: 0x3b82f6, roughness: 0.5 })
    const cube = new THREE.Mesh(new THREE.BoxGeometry(1.2, 1.2, 1.2), material)
    cube.position.y = 0.6
    cube.castShadow = cube.receiveShadow = true
    this.scene.add(cube)

    // ④ 注册更新逻辑（闭包绑定 cube）；主循环只消费 updatables
    this.updatables.push({
      update(dt) {
        cube.rotation.y += dt * params.speed
      }
    })
    return { material, cube }
  }
}

/* ============ Loop 层：时钟驱动 update 链 + render ============ */
class Loop {
  constructor(stage, updatables) {
    this.stage = stage
    this.updatables = updatables
    this.clock = new THREE.Timer()
    this.clock.connect(document) // r183 起 Clock 已弃用，改用 Timer
  }
  start() {
    this.stage.renderer.setAnimationLoop(() => this.tick())
  }
  tick() {
    this.clock.update() // 每帧先 update，再读取
    const dt = this.clock.getDelta()        // 距上一帧的秒数，与帧率无关
    for (const u of this.updatables) u.update(dt)
    this.stage.controls.update()
    this.stage.renderer.render(this.stage.scene, this.stage.camera)
  }
}

/* ============ App 层：编排 ============ */
const params = { speed: 1 }
const stage = new Stage(document.getElementById('app'))
const builder = new WorldBuilder(stage.scene)
const { material, cube } = builder.build()
new Loop(stage, builder.updatables).start()

/* ---- lil-gui 参数面板 ---- */
const gui = new GUI({ title: '调试面板' })
gui.add(params, 'speed', 0, 5, 0.1).name('旋转速度')
gui.add(material, 'wireframe').name('线框模式')
gui.addColor({ c: '#3b82f6' }, 'c').name('颜色').onChange(v => material.color.set(v))
gui.add(cube, 'visible').name('是否显示')

/* ---- FPS / draw call 统计（每 0.5s 采样，避免数字抖动）---- */
let frames = 0
let last = performance.now()
stage.renderer.setAnimationLoop(() => {
  frames++
  const now = performance.now()
  if (now - last >= 500) {
    const fps = Math.round((frames * 1000) / (now - last))
    const info = stage.renderer.info.render
    console.log(`FPS ${fps} | calls ${info.calls} | triangles ${info.triangles}`)
    frames = 0
    last = now
  }
})

常见问题解答 ​

Q1：坐标轴颜色和坐标系怎么对应？

Three.js 是右手坐标系：拇指 +X（红）、食指 +Y（绿）、中指 +Z（蓝）。注意 +Z 朝向屏幕外，与 Unity 的左手系相反。口诀：红右、绿上、蓝前。

Q2：为什么 FPS 不每帧算而要 0.5s 采样一次？

每帧实时算会显示 58、62、57、63 这种抖动数字，视觉上反而像卡顿。0.5s 窗口的平均值更稳定、更贴近真实体感。

Q3：lil-gui 改参数什么时候要 material.needsUpdate = true？

改 roughness / metalness / color / opacity / wireframe 这类标量属性不需要；改 map / normalMap / emissiveMap 等贴图槽位或 side、vertexColors 这类影响着色器编译的开关时才需要。

动手小练习 ​

  1. 再加一盏 PointLight 放在 (2,2,2)，并用一个自发光小球标记光源位置。
  2. 把阴影 mapSize 调到 4096，观察阴影边缘清晰度变化与帧率影响。
  3. 用 Group 包两个 mesh，旋转 Group，体会"父节点变换带动子节点"。
