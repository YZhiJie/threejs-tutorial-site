7.3 实战案例：流动贴图与程序化纹理动画 ​

核心概念讲解 ​

很多"动态材质"不需要换图、不需要动几何体，只需要让纹理坐标随时间变化或每帧重绘 Canvas。本篇综合三种常见手法：

  1. offset 流动：整张图沿 UV 平移（传送带、河流）
  2. 双层流动 + 混合：两张速度/方向不同的纹理叠加，做更丰富的流动（岩浆、能量场）
  3. Canvas 实时重绘：直接在 2D 画布上画动态内容（雷达扫描、数据屏），needsUpdate 上传

选型建议：能用 UV 流动就别整图重绘（便宜得多）；需要程序化复杂图形（扫描线、波形文字）时才用 CanvasTexture 重绘。

纹理旋转中心 ​

texture.center.set(0.5, 0.5) 把旋转/缩放中心设到 UV 中点，再改 texture.rotation 可让贴图绕中心旋转，做雷达转盘很合适。

代码示例 ​

一个"科幻能量地板"：底层网格纹理缓慢滚动 + 顶层扫描纹理反向流动 + 一个旋转的 Canvas 雷达：

js
import * as THREE from 'three'
import { OrbitControls } from 'three/addons/controls/OrbitControls.js'
import GUI from 'three/addons/libs/lil-gui.module.min.js'

/* ---- 程序化纹理 1：网格 ---- */
function gridCanvas() {
  const cv = document.createElement('canvas')
  cv.width = cv.height = 256
  const ctx = cv.getContext('2d')
  ctx.fillStyle = '#0a0f1a'
  ctx.fillRect(0, 0, 256, 256)
  ctx.strokeStyle = '#1d4ed8'
  ctx.lineWidth = 2
  for (let i = 0; i <= 8; i++) {
    const p = i * 32
    ctx.beginPath(); ctx.moveTo(p, 0); ctx.lineTo(p, 256); ctx.stroke()
    ctx.beginPath(); ctx.moveTo(0, p); ctx.lineTo(256, p); ctx.stroke()
  }
  return cv
}
/* ---- 程序化纹理 2：扫描虚线（透明背景，用来叠加）---- */
function scanCanvas() {
  const cv = document.createElement('canvas')
  cv.width = cv.height = 256
  const ctx = cv.getContext('2d')
  ctx.strokeStyle = '#38bdf8'
  ctx.lineWidth = 3
  ctx.setLineDash([18, 14])
  for (let i = 0; i < 6; i++) {
    ctx.beginPath()
    ctx.moveTo(-40, i * 52)
    ctx.lineTo(296, i * 52 - 60)
    ctx.stroke()
  }
  return cv
}

const grid = new THREE.CanvasTexture(gridCanvas())
grid.colorSpace = THREE.SRGBColorSpace
grid.wrapS = grid.wrapT = THREE.RepeatWrapping
grid.repeat.set(6, 6)

const scan = new THREE.CanvasTexture(scanCanvas())
scan.colorSpace = THREE.SRGBColorSpace
scan.wrapS = scan.wrapT = THREE.RepeatWrapping
scan.repeat.set(6, 6)

const scene = new THREE.Scene()
scene.background = new THREE.Color(0x05070c)
const camera = new THREE.PerspectiveCamera(50, innerWidth / innerHeight, 0.1, 100)
camera.position.set(0, 4.2, 6.4)
const renderer = new THREE.WebGLRenderer({ antialias: true })
renderer.setSize(innerWidth, innerHeight)
renderer.setPixelRatio(Math.min(devicePixelRatio, 2))
document.getElementById('app').appendChild(renderer.domElement)
const controls = new OrbitControls(camera, renderer.domElement)
controls.enableDamping = true
controls.maxPolarAngle = Math.PI / 2 - 0.05

scene.add(new THREE.HemisphereLight(0x88aaff, 0x101420, 0.7))

const floorMat = new THREE.MeshStandardMaterial({
  map: grid,
  emissive: new THREE.Color(0x0b3a66),
  emissiveMap: scan,       // 扫描纹理驱动自发光
  emissiveIntensity: 1.4,
  roughness: 0.5,
  metalness: 0.3
})
const floor = new THREE.Mesh(new THREE.PlaneGeometry(12, 12), floorMat)
floor.rotation.x = -Math.PI / 2
scene.add(floor)

/* ---- CanvasTexture 雷达屏（每帧重绘）---- */
const radarCv = document.createElement('canvas')
radarCv.width = radarCv.height = 256
const rctx = radarCv.getContext('2d')
const radarTex = new THREE.CanvasTexture(radarCv)
radarTex.colorSpace = THREE.SRGBColorSpace
const screen = new THREE.Mesh(
  new THREE.PlaneGeometry(2.2, 2.2),
  new THREE.MeshBasicMaterial({ map: radarTex, transparent: true })
)
screen.position.set(0, 1.3, -3.2)
scene.add(screen)

function drawRadar(t) {
  rctx.clearRect(0, 0, 256, 256)
  rctx.fillStyle = 'rgba(2,12,20,0.92)'
  rctx.fillRect(0, 0, 256, 256)
  // 同心圆
  rctx.strokeStyle = 'rgba(56,189,248,.55)'
  for (let r = 32; r <= 112; r += 32) {
    rctx.beginPath(); rctx.arc(128, 128, r, 0, Math.PI * 2); rctx.stroke()
  }
  // 十字线
  rctx.beginPath(); rctx.moveTo(128, 8); rctx.lineTo(128, 248); rctx.stroke()
  rctx.beginPath(); rctx.moveTo(8, 128); rctx.lineTo(248, 128); rctx.stroke()
  // 旋转扫描扇区
  const a = t * 1.6
  const grad = rctx.createConicGradient ? null : null
  rctx.save()
  rctx.translate(128, 128)
  rctx.rotate(a)
  const g = rctx.createLinearGradient(0, 0, 112, 0)
  g.addColorStop(0, 'rgba(56,189,248,.8)')
  g.addColorStop(1, 'rgba(56,189,248,0)')
  rctx.fillStyle = g
  rctx.beginPath(); rctx.moveTo(0, 0); rctx.arc(0, 0, 112, -0.35, 0); rctx.closePath(); rctx.fill()
  rctx.restore()
  radarTex.needsUpdate = true   // 通知重新上传
}

const params = { gridSpeed: 0.12, scanSpeed: 0.45, emissive: 1.4 }
const gui = new GUI({ title: '流动纹理' })
gui.add(params, 'gridSpeed', 0, 1, 0.02).name('网格滚动')
gui.add(params, 'scanSpeed', 0, 1.5, 0.02).name('扫描流速')
gui.add(params, 'emissive', 0, 3, 0.05).name('自发光强度').onChange(v => floorMat.emissiveIntensity = v)

const clock = new THREE.Timer()

clock.connect(document) // r183 起 Clock 已弃用，改用 Timer
renderer.setAnimationLoop(() => {
  clock.update() // 每帧先 update，再读取
  const dt = clock.getDelta()
  const t = clock.getElapsed()
  // 两层纹理反向、不同速度流动
  grid.offset.y += params.gridSpeed * dt
  scan.offset.y -= params.scanSpeed * dt
  scan.offset.x += params.scanSpeed * dt * 0.4
  drawRadar(t)
  controls.update()
  renderer.render(scene, camera)
})

常见问题解答 ​

Q1：emissiveMap 和 map 同时用，颜色怎么叠加？

map 决定基础反照率（受光），emissiveMap 决定哪些区域自发光（不受光、恒定亮）。二者相乘/叠加后，自发光区域即使在暗处也亮，非常适合做"灯带、屏幕、魔法纹路"。

Q2：每帧重绘 Canvas 会卡吗？

256×256 的简单 2D 绘制开销很小；但 1024×1024 全画面每帧重绘 + 上传会有明显成本。优化方向：减小画布、只在内容变化时 needsUpdate、或改用真实的动态视频/渲染目标（RenderTarget）。

Q3：两张纹理共享同一个 offset 吗？

不共享。offset/repeat/wrap 是每个 Texture 对象独立的状态。map 和 emissiveMap 如果是两张不同的 Texture，需要分别设置；若想同步流动，让它们引用同一个 texture 对象即可。

动手小练习 ​

  1. 把网格纹理换成水面波纹（同心圆环），用 offset + rotation 模拟漩涡。
  2. 给雷达加两个随机移动的"目标点"，扫描扇区扫过时点变亮。
  3. 用滚动纹理做一条传送带，并在上面放几个跟随纹理速度移动的箱子。
